import { useEffect, useState } from 'react';
import { useParams, Link, useNavigate } from 'react-router-dom';
import { useQuery } from '@tanstack/react-query';
import { Play, ArrowLeft, Plus, Check, Calendar, Clock, Star, Tv, Film as FilmIcon, Lock } from 'lucide-react';
import { tmdb, img, titleName, titleYear, pickTrailer } from '@/lib/tmdb';
import { useAuth, hasActiveSubscription } from '@/context/AuthContext';
import { useFavorite } from '@/hooks/useFavorite';
import { useUpsertHistory } from '@/hooks/useWatchHistory';
import { SubscriptionGate } from '@/components/SubscriptionGate';
import { PosterCard } from '@/components/cards/PosterCard';
import { Carousel } from '@/components/Carousel';
import { FullScreenLoader } from '@/components/ui/Feedback';
import type { MediaType } from '@/types';

export function TitleDetailPage() {
  const { type, id } = useParams<{ type: MediaType; id: string }>();
  const navigate = useNavigate();
  const { user, subscription } = useAuth();
  const mediaType = (type === 'tv' ? 'tv' : 'movie') as MediaType;
  const tmdbId = Number(id);

  const detail = useQuery({
    queryKey: ['detail', mediaType, tmdbId],
    queryFn: () => tmdb.details(mediaType, tmdbId),
    enabled: !!tmdbId,
  });

  const { isFavorite, toggle } = useFavorite(tmdbId, mediaType);
  const upsertHistory = useUpsertHistory();
  const [gateOpen, setGateOpen] = useState(false);
  const [playing, setPlaying] = useState(false);

  const active = hasActiveSubscription(subscription);

  useEffect(() => {
    if (detail.data) {
      upsertHistory.mutate({
        title: detail.data,
        positionSeconds: 0,
        durationSeconds: (detail.data.runtime ?? detail.data.episode_run_time?.[0] ?? 0) * 60,
      });
    }
  }, [detail.data]); // eslint-disable-line react-hooks/exhaustive-deps

  if (detail.isLoading) return <FullScreenLoader label="Carregando título…" />;
  if (!detail.data) {
    return (
      <div className="container-app flex min-h-[60vh] flex-col items-center justify-center gap-4 text-ink-300">
        <p>Título não encontrado.</p>
        <Link to="/" className="btn-primary">Voltar ao início</Link>
      </div>
    );
  }

  const t = detail.data;
  const backdrop = img(t.backdrop_path, 'original');
  const poster = img(t.poster_path, 'w500');
  const trailer = pickTrailer((t as { videos?: { results: [] } }).videos?.results as never);
  const genres = t.genres ?? [];
  const runtime = t.runtime ?? t.episode_run_time?.[0];
  const similar = ((t as { similar?: { results: TmdbLike[] } }).similar?.results ?? []).filter((r) => r.poster_path).slice(0, 12);

  function handlePlay() {
    if (!active) {
      setGateOpen(true);
      return;
    }
    if (!user) {
      navigate('/login');
      return;
    }
    setPlaying(true);
  }

  return (
    <div>
      <div className="relative h-[50vh] min-h-[360px] w-full overflow-hidden sm:h-[60vh]">
        {backdrop && <img src={backdrop} alt="" className="absolute inset-0 h-full w-full object-cover" />}
        <div className="absolute inset-0 bg-gradient-to-t from-ink-950 via-ink-950/50 to-transparent" />
        <div className="absolute inset-0 bg-gradient-to-r from-ink-950/80 via-transparent to-transparent" />
      </div>

      <div className="container-app -mt-40 sm:-mt-48">
        <Link
          to="/"
          className="mb-4 inline-flex items-center gap-1.5 text-sm text-ink-300 transition hover:text-white"
        >
          <ArrowLeft className="h-4 w-4" /> Voltar
        </Link>

        <div className="flex flex-col gap-6 sm:flex-row">
          <div className="mx-auto w-40 flex-shrink-0 sm:mx-0 sm:w-56">
            <div className="overflow-hidden rounded-2xl border border-white/10 shadow-2xl">
              {poster ? (
                <img src={poster} alt={titleName(t)} className="w-full" />
              ) : (
                <div className="aspect-[2/3] bg-ink-800" />
              )}
            </div>
          </div>

          <div className="flex-1">
            <div className="flex items-center gap-2 text-xs text-ink-300">
              {mediaType === 'tv' ? (
                <span className="inline-flex items-center gap-1"><Tv className="h-3.5 w-3.5" /> Série</span>
              ) : (
                <span className="inline-flex items-center gap-1"><FilmIcon className="h-3.5 w-3.5" /> Filme</span>
              )}
            </div>
            <h1 className="mt-1 font-display text-3xl tracking-wide text-white sm:text-5xl">{titleName(t)}</h1>

            <div className="mt-3 flex flex-wrap items-center gap-x-4 gap-y-2 text-sm text-ink-200">
              {t.vote_average > 0 && (
                <span className="inline-flex items-center gap-1 font-semibold text-amber-400">
                  <Star className="h-4 w-4 fill-amber-400" /> {t.vote_average.toFixed(1)}
                </span>
              )}
              {titleYear(t) && (
                <span className="inline-flex items-center gap-1"><Calendar className="h-4 w-4" /> {titleYear(t)}</span>
              )}
              {runtime ? (
                <span className="inline-flex items-center gap-1"><Clock className="h-4 w-4" /> {runtime} min</span>
              ) : null}
              {mediaType === 'tv' && t.number_of_seasons ? (
                <span>{t.number_of_seasons} temporada(s)</span>
              ) : null}
            </div>

            {genres.length > 0 && (
              <div className="mt-3 flex flex-wrap gap-2">
                {genres.map((g) => (
                  <span key={g.id} className="chip">{g.name}</span>
                ))}
              </div>
            )}

            {t.overview && <p className="mt-4 max-w-2xl text-sm text-ink-200 sm:text-base">{t.overview}</p>}

            <div className="mt-6 flex flex-wrap items-center gap-3">
              <button onClick={handlePlay} className="btn-primary">
                <Play className="h-4 w-4 fill-white" /> {active ? 'Assistir agora' : 'Assinar para assistir'}
              </button>
              {user && (
                <button onClick={() => toggle()} className="btn-outline">
                  {isFavorite ? <Check className="h-4 w-4" /> : <Plus className="h-4 w-4" />}
                  {isFavorite ? 'Favoritado' : 'Favoritar'}
                </button>
              )}
            </div>
          </div>
        </div>

        {playing && active && (
          <div className="mt-8 overflow-hidden rounded-2xl border border-white/10 bg-black">
            <div className="aspect-video w-full">
              {trailer ? (
                <iframe
                  className="h-full w-full"
                  src={`https://www.youtube.com/embed/${trailer.key}?autoplay=1&rel=0`}
                  title="Trailer"
                  allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture"
                  allowFullScreen
                />
              ) : (
                <div className="flex h-full w-full items-center justify-center text-ink-400">
                  Trailer não disponível para este título.
                </div>
              )}
            </div>
          </div>
        )}

        {!active && (
          <div className="mt-8 flex items-center gap-3 rounded-2xl border border-brand-600/30 bg-brand-900/20 p-4">
            <Lock className="h-5 w-5 text-brand-400" />
            <p className="flex-1 text-sm text-ink-200">
              Este é um conteúdo exclusivo. Assine um plano para assistir agora.
            </p>
            <Link to="/minha-assinatura" className="btn-primary text-sm">Assinar</Link>
          </div>
        )}

        {similar.length > 0 && (
          <Carousel title="Títulos semelhantes" className="mt-12">
            {similar.map((s) => (
              <PosterCard
                key={s.id}
                title={{ ...s, media_type: mediaType } as never}
                forceType={mediaType}
              />
            ))}
          </Carousel>
        )}
      </div>

      <SubscriptionGate open={gateOpen} onClose={() => setGateOpen(false)} />
    </div>
  );
}

type TmdbLike = {
  id: number;
  title?: string;
  name?: string;
  poster_path: string | null;
  backdrop_path: string | null;
  vote_average: number;
  overview: string;
  release_date?: string;
  first_air_date?: string;
};
